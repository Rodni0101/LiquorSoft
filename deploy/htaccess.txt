# LiquorSoft - Apache (InfinityFree, GoogieHost, Byet, cPanel/DirectAdmin)
# Se sube como .htaccess en la raíz de htdocs/public_html.

Options -Indexes
DirectoryIndex index.html

# Bloquear archivos sensibles aunque se suban por error.
<FilesMatch "(^\.env|^\.git|^\.htaccess|^\.installed|\.sql$|\.md$|composer\.(json|lock)$)">
  Require all denied
</FilesMatch>

# El frontend es una SPA de Angular: cualquier ruta que no exista en disco
# debe devolver index.html. /api/*.php son archivos reales y no se reescriben.
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  RewriteRule ^api/ - [L]

  RewriteCond %{REQUEST_FILENAME} -f [OR]
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^ - [L]

  RewriteRule ^ index.html [L]
</IfModule>


<IfModule mod_headers.c>
  Header unset X-Powered-By
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "DENY"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"
  Header always set Cross-Origin-Opener-Policy "same-origin"
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
</IfModule>

<IfModule mod_headers.c>
  # index.html no se cachea para que los despliegues se vean de inmediato.
  <FilesMatch "^index\.html$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
  </FilesMatch>

  # Los nombres de los bundles de Angular llevan hash, se pueden cachear.
  <FilesMatch "\.(js|css|svg|png|jpe?g|webp|ico|woff2?)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

# Mensajes genéricos: no revelar rutas internas ni la versión de PHP.
<IfModule mod_php.c>
  php_flag display_errors Off
  php_flag log_errors On
</IfModule>
